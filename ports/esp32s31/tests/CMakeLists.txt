include(${port}/opus.cmake)
# Use the component's exact source list and fixed-point flags on the host too.
function(idf_component_register)
  cmake_parse_arguments(OPUS "" "" "SRCS;INCLUDE_DIRS;PRIV_INCLUDE_DIRS" ${ARGN})
  add_library(s31-opus STATIC ${OPUS_SRCS})
  target_include_directories(s31-opus PUBLIC ${OPUS_INCLUDE_DIRS} PRIVATE ${OPUS_PRIV_INCLUDE_DIRS})
  set(COMPONENT_LIB s31-opus PARENT_SCOPE)
endfunction()
set(IDF_TARGET host)
add_subdirectory(${ink_opus_source} opus-build)
get_target_property(ink_opus_host_flags s31-opus COMPILE_OPTIONS)
list(REMOVE_ITEM ink_opus_host_flags -Wall -Wextra -Werror)
set_property(TARGET s31-opus PROPERTY COMPILE_OPTIONS ${ink_opus_host_flags})
target_compile_options(s31-opus PRIVATE -Wno-unknown-warning-option)
