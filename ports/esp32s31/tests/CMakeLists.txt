cmake_minimum_required(VERSION 3.16)
project(s31_io_tests C)
set(CMAKE_C_STANDARD 11)
option(INK_SANITIZE "Address and undefined behavior sanitizers" OFF)
set(port ${CMAKE_CURRENT_LIST_DIR}/..)
set(root ${port}/../..)
add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
if(INK_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include_directories(${port} ${root}/platform/include ${root}/drivers/chips ${root}/boards)
add_executable(test-s31-input logic.c ${root}/boards/metalio_plus/metalio_touch.c)
add_executable(test-s31-power power.c ${port}/hal_power_s31.c)
add_executable(test-s31-board board.c ${root}/boards/metalio_plus/metalio_board.c
  ${root}/boards/metalio_plus/metalio_epd_power.c ${root}/drivers/chips/tca9555/tca9555.c
  ${root}/drivers/chips/tps65185/tps65185.c)
target_include_directories(test-s31-power PRIVATE stub)
target_compile_definitions(test-s31-power PRIVATE INK_S31_LIGHT_SLEEP=1)
add_executable(test-s31-awake power.c ${port}/hal_power_s31.c)
target_include_directories(test-s31-awake PRIVATE stub)
target_compile_definitions(test-s31-awake PRIVATE INK_S31_LIGHT_SLEEP=0)
add_executable(test-s31-boot boot.c ${port}/hal_device_s31.c ${port}/main/main.c)
target_include_directories(test-s31-boot PRIVATE boot_stub)
add_executable(test-s31-panic panic.c ${port}/s31_panic.c)
target_include_directories(test-s31-panic PRIVATE panic_stub boot_stub)
add_executable(test-s31-stubs stubs.c ${port}/hal_stubs.c)
target_include_directories(test-s31-stubs PRIVATE stub)
target_compile_options(test-s31-stubs PRIVATE -Wno-unused-parameter)
find_package(Threads REQUIRED)
add_executable(test-s31-display display.c ${root}/drivers/display/pqbn/pqbn_v4.c)
target_include_directories(test-s31-display PRIVATE display_stub ${root}/drivers/display/pqbn)
target_link_libraries(test-s31-display PRIVATE Threads::Threads m)
enable_testing()
add_test(NAME unit-s31-input COMMAND test-s31-input)
add_test(NAME unit-s31-power COMMAND test-s31-power)
add_test(NAME unit-s31-awake COMMAND test-s31-awake)
add_test(NAME unit-s31-boot COMMAND test-s31-boot)
add_test(NAME unit-s31-panic COMMAND test-s31-panic)
add_test(NAME unit-s31-stubs COMMAND test-s31-stubs)
add_test(NAME unit-s31-board COMMAND test-s31-board)
foreach(case regions timeout guard lines wake-overlap shutdown panic)
  add_test(NAME unit-s31-display-${case} COMMAND test-s31-display ${case})
endforeach()
