function redirectOnLoad() {
if (this.SfdcApp && this.SfdcApp.projectOneNavigator) { SfdcApp.projectOneNavigator.handleRedirect('https://support.remarkable.com/s//articles/Knowledge/How-to-use-continuous-pages'); }  else
if (window.location.replace){
window.location.replace('https://support.remarkable.com/s//articles/Knowledge/How-to-use-continuous-pages');
} else {
window.location.href ='https://support.remarkable.com/s//articles/Knowledge/How-to-use-continuous-pages';
}
}
redirectOnLoad();
