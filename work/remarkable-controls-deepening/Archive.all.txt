function redirectOnLoad() {
if (this.SfdcApp && this.SfdcApp.projectOneNavigator) { SfdcApp.projectOneNavigator.handleRedirect('https://support.remarkable.com/s//articles/Knowledge/Archive'); }  else
if (window.location.replace){
window.location.replace('https://support.remarkable.com/s//articles/Knowledge/Archive');
} else {
window.location.href ='https://support.remarkable.com/s//articles/Knowledge/Archive';
}
}
redirectOnLoad();
