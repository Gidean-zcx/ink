set(spike ${CMAKE_CURRENT_LIST_DIR})
set(INK_SPIKE_LUA "${CMAKE_SOURCE_DIR}/work/appfw-spike/lua-5.4.8" CACHE PATH "Unmodified Lua 5.4.8 source")
if(NOT EXISTS "${INK_SPIKE_LUA}/src/lua.h")
  message(FATAL_ERROR "Run tools/appfw-spike/run.sh fetch first")
endif()
file(GLOB lua_sources CONFIGURE_DEPENDS "${INK_SPIKE_LUA}/src/*.c")
list(FILTER lua_sources EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(appfw_lua_vm STATIC EXCLUDE_FROM_ALL ${lua_sources})
target_include_directories(appfw_lua_vm SYSTEM PUBLIC "${INK_SPIKE_LUA}/src")
target_link_libraries(appfw_lua_vm PUBLIC m)
include(${spike}/mquickjs.cmake)
foreach(candidate c json lua mquickjs)
  add_executable(appfw-${candidate} EXCLUDE_FROM_ALL
    ${CMAKE_SOURCE_DIR}/simulator/runtime/sim_main.c
    ${spike}/harness.c ${spike}/bridge.c ${spike}/candidate_${candidate}.c)
  target_include_directories(appfw-${candidate} PRIVATE ${spike})
  target_link_libraries(appfw-${candidate} PRIVATE ink_apps ink_system ink_port_sim)
  target_compile_options(appfw-${candidate} PRIVATE ${INK_WARN})
  target_compile_definitions(appfw-${candidate} PRIVATE INK_SPIKE_KIND="${candidate}")
  set_target_properties(appfw-${candidate} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/appfw-spike")
  add_dependencies(appfw-${candidate} ink_fonts ink_ui_fonts ink_icons ink_test_content)
endforeach()
target_link_libraries(appfw-lua PRIVATE appfw_lua_vm)
target_link_libraries(appfw-mquickjs PRIVATE appfw_mquickjs_vm)
