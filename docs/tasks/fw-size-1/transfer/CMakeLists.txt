cmake_minimum_required(VERSION 3.22)
project(fw_size_transfer C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(INK_SANITIZE "Address and undefined behavior sanitizers" OFF)
get_filename_component(root ${CMAKE_CURRENT_LIST_DIR}/../../../.. ABSOLUTE)
set(ap ${root}/ports/esp32s31/vendor/ap_transfer)
set(tests ${root}/ports/esp32s31/tests)
add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
if(INK_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(shared_miniz STATIC ${root}/third_party/reader_miniz/miniz.c)
target_include_directories(shared_miniz SYSTEM PUBLIC ${root}/third_party ${root}/apps/reader/vendor/shim)
target_compile_definitions(shared_miniz PUBLIC MINIZ_NO_ARCHIVE_APIS MINIZ_NO_STDIO MINIZ_NO_TIME MINIZ_NO_DEFLATE_APIS
  tinfl_decompress=ink_tinfl_decompress tinfl_decompress_mem_to_heap=ink_tinfl_decompress_mem_to_heap
  tinfl_decompress_mem_to_mem=ink_tinfl_decompress_mem_to_mem tinfl_decompress_mem_to_callback=ink_tinfl_decompress_mem_to_callback
  mz_crc32=ink_mz_crc32 mz_adler32=ink_mz_adler32 mz_free=ink_mz_free)
target_compile_options(shared_miniz PRIVATE -w)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/fixtures.py ${CMAKE_BINARY_DIR}/fixtures
  COMMAND_ERROR_IS_FATAL ANY)
foreach(engine zlib miniz)
  add_executable(inflate-${engine} inflate.cxx)
  add_executable(transfer-${engine} ${tests}/transfer.cxx ${tests}/transfer_policy.c ${ap}/ap_transfer.cc
    ${root}/third_party/cjson/cJSON.c ${root}/system/util/ink_json.c)
  target_include_directories(transfer-${engine} PRIVATE ${tests}/transfer_stub ${ap}
    ${root}/third_party/cjson ${root}/system/include ${root}/platform/include ${root}/third_party/jsmn)
  if(engine STREQUAL zlib)
    add_library(reference_zlib STATIC ${ap}/zlib/adler32.c ${ap}/zlib/crc32.c ${ap}/zlib/inffast.c
      ${ap}/zlib/inflate.c ${ap}/zlib/inftrees.c ${ap}/zlib/zutil.c)
    target_include_directories(reference_zlib PUBLIC ${ap}/zlib)
    target_link_libraries(transfer-${engine} PRIVATE reference_zlib)
    target_link_libraries(inflate-${engine} PRIVATE reference_zlib)
  else()
    target_link_libraries(transfer-${engine} PRIVATE shared_miniz)
    target_link_libraries(inflate-${engine} PRIVATE shared_miniz)
  endif()
  target_link_libraries(transfer-${engine} PRIVATE m)
  add_test(NAME transfer-${engine} COMMAND transfer-${engine})
  add_test(NAME inflate-${engine} COMMAND inflate-${engine} ${CMAKE_BINARY_DIR}/fixtures)
endforeach()
if(APPLE)
  set_source_files_properties(${root}/third_party/cjson/cJSON.c PROPERTIES COMPILE_OPTIONS -Wno-error=deprecated-declarations)
endif()
